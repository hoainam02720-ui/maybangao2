// ==================================== // 🌾 NAMRICE AUTO - REAL SERVER // ==================================== const express = require("express") const axios = require("axios") const cors = require("cors") const http = require("http") const { Server } = require("socket.io") const app = express() const server = http.createServer(app) const io = new Server(server, { cors: { origin: "*" } }) app.use(cors()) app.use(express.json()) app.use(express.static("public")) // ==================================== // ⚙️ CONFIG // ==================================== const PORT = process.env.PORT || 3000 const MACHINE = { name: "Máy bán gạo A4", address: "Đường A4 Cần Thơ", lat: 10.0452, lon: 105.7469 } const SYSTEM_FEE = 0.165 // ==================================== // 💰 PRICE // ==================================== const PRICE = { rice: { st25: 180000, jasmine: 220000 }, ride: { bike: 8000, car4: 13000, car7: 18000 }, delivery: { bike: 8000, truck500: 13000, truck1t: 18000, truck25: 22000, truck35: 26000 } } // ==================================== // 🗃 DATABASE // ==================================== let orders = [] let shippers = [] // ==================================== // 🗺 GET LATLON // ==================================== async function getLatLon(address){ const url = `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(address)}` const res = await axios.get(url) if(!res.data[0]){ throw new Error("Address not found") } return { lat: parseFloat(res.data[0].lat), lon: parseFloat(res.data[0].lon) } } // ==================================== // 🗺 GET REAL ROUTE // ==================================== async function getRoute( lat1, lon1, lat2, lon2 ){ const url = `https://router.project-osrm.org/route/v1/driving/${lon1},${lat1};${lon2},${lat2}?overview=full&geometries=geojson` const res = await axios.get(url) return res.data.routes[0] } // ==================================== // 🌾 CREATE RICE ORDER // ==================================== app.post("/api/order/rice", async(req,res)=>{ try{ const { riceType, bags, customerAddress } = req.body const customer = await getLatLon(customerAddress) const route = await getRoute( MACHINE.lat, MACHINE.lon, customer.lat, customer.lon ) const km = route.distance / 1000 const shippingFee = km * PRICE.ride.bike const riceFee = bags * PRICE.rice[riceType] const total = Math.round(shippingFee + riceFee) const order = { id: Date.now(), type: "rice", riceType, bags, totalKg: bags * 5, machine: MACHINE, customerAddress, customer, route, km, shippingFee, riceFee, total, status: "waiting_shipper", createdAt: new Date() } orders.push(order) io.emit("new_order", order) res.json({ success:true, order }) }catch(err){ res.status(500).json({ success:false, message: err.message }) } }) // ==================================== // 🚕 CREATE RIDE ORDER // ==================================== app.post("/api/order/ride", async(req,res)=>{ try{ const { pickup, destination, vehicle } = req.body const pickupLocation = await getLatLon(pickup) const destinationLocation = await getLatLon(destination) const route = await getRoute( pickupLocation.lat, pickupLocation.lon, destinationLocation.lat, destinationLocation.lon ) const km = route.distance / 1000 const total = Math.round( km * PRICE.ride[vehicle] ) const order = { id: Date.now(), type:"ride", pickup, destination, vehicle, pickupLocation, destinationLocation, route, km, total, status:"waiting_shipper", createdAt:new Date() } orders.push(order) io.emit("new_order", order) res.json({ success:true, order }) }catch(err){ res.status(500).json({ success:false, message: err.message }) } }) // ==================================== // 📦 CREATE DELIVERY ORDER // ==================================== app.post("/api/order/delivery", async(req,res)=>{ try{ const { pickup, destination, vehicle } = req.body const pickupLocation = await getLatLon(pickup) const destinationLocation = await getLatLon(destination) const route = await getRoute( pickupLocation.lat, pickupLocation.lon, destinationLocation.lat, destinationLocation.lon ) const km = route.distance / 1000 const total = Math.round( km * PRICE.delivery[vehicle] ) const order = { id: Date.now(), type:"delivery", pickup, destination, vehicle, pickupLocation, destinationLocation, route, km, total, status:"waiting_shipper", createdAt:new Date() } orders.push(order) io.emit("new_order", order) res.json({ success:true, order }) }catch(err){ res.status(500).json({ success:false, message: err.message }) } }) // ==================================== // 🛵 SHIPPER RECEIVE ORDER // ==================================== app.post("/api/shipper/receive", (req,res)=>{ const { orderId, shipperId } = req.body const order = orders.find( o => o.id == orderId ) if(!order){ return res.json({ success:false }) } order.status = "shipper_received" order.shipperId = shipperId io.emit( "order_updated", order ) res.json({ success:true, order }) }) // ==================================== // 📍 SHIPPER ARRIVED // ==================================== app.post("/api/shipper/arrived", (req,res)=>{ const {orderId} = req.body const order = orders.find( o => o.id == orderId ) if(!order){ return res.json({ success:false }) } order.status = "arrived" io.emit( "order_updated", order ) res.json({ success:true, order }) }) // ==================================== // 🌾 OPEN ESP32 // ==================================== app.post("/api/machine/open", async(req,res)=>{ try{ await axios.get( "http://192.168.1.14/mo" ) res.json({ success:true }) }catch(err){ res.json({ success:false, message: "ESP32 offline" }) } }) // ==================================== // ✅ COMPLETE ORDER // ==================================== app.post("/api/order/complete", (req,res)=>{ const {orderId} = req.body const order = orders.find( o => o.id == orderId ) if(!order){ return res.json({ success:false }) } const fee = Math.round( order.total * SYSTEM_FEE ) const shipperReceive = order.total - fee order.status = "completed" order.systemFee = fee order.shipperReceive = shipperReceive io.emit( "order_completed", order ) res.json({ success:true, customerPaid: order.total, systemFee:fee, shipperReceive }) }) // ==================================== // 📦 GET ORDERS // ==================================== app.get("/api/orders",(req,res)=>{ res.json(orders) }) // ==================================== // 🔌 SOCKET // ==================================== io.on("connection",(socket)=>{ console.log( "🟢 Client connected" ) socket.on( "shipper_location", (data)=>{ io.emit( "shipper_location_update", data ) } ) socket.on( "disconnect", ()=>{ console.log( "🔴 Client disconnected" ) } ) }) // ==================================== // 🚀 START // ==================================== server.listen(PORT,()=>{ console.log( `🌾 NamRice Auto Running Port ${PORT}` ) })
