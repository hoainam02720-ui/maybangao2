/* ========================= ĐỔI LOẠI XE ========================= */ function selectVehicle(type){ selectedVehicle = type; alert("Đã chọn: " + type); } /* ========================= GPS REALTIME ========================= */ let userMarker; /* ========================= THEO DÕI GPS LIÊN TỤC ========================= */ navigator.geolocation.watchPosition( function(position){ let lat = position.coords.latitude; let lng = position.coords.longitude; /* ========================= XOÁ MARKER CŨ ========================= */ if(userMarker){ map.removeLayer(userMarker); } /* ========================= TẠO MŨI TÊN ========================= */ let arrowIcon = L.divIcon({ html:'⬆️', className:'', iconSize:[30,30] }); /* ========================= HIỆN VỊ TRÍ HIỆN TẠI ========================= */ userMarker = L.marker( [lat,lng], { icon:arrowIcon } ).addTo(map); /* ========================= MAP ĐI THEO MÌNH ========================= */ map.setView([lat,lng]); } ); /* ========================= TÍNH KHOẢNG CÁCH ========================= */ async function calculate(){ let from = document.getElementById("from").value + ", Việt Nam"; let to = document.getElementById("to").value + ", Việt Nam"; /* ========================= TÌM TOẠ ĐỘ ========================= */ let fromData = await fetch( 'https://nominatim.openstreetmap.org/search?format=json&q=' + encodeURIComponent(from) ); let fromJson = await fromData.json(); let toData = await fetch( 'https://nominatim.openstreetmap.org/search?format=json&q=' + encodeURIComponent(to) ); let toJson = await toData.json(); /* ========================= KIỂM TRA ========================= */ if(fromJson.length==0 || toJson.length==0){ alert("Không tìm thấy địa điểm"); return; } let lat1 = fromJson[0].lat; let lon1 = fromJson[0].lon; let lat2 = toJson[0].lat; let lon2 = toJson[0].lon; /* ========================= ROUTING ========================= */ let routeData = await fetch( `https://router.project-osrm.org/route/v1/driving/${lon1},${lat1};${lon2},${lat2}?overview=full&geometries=geojson` ); let routeJson = await routeData.json(); /* ========================= KHOẢNG CÁCH ========================= */ let distance = routeJson.routes[0].distance / 1000; /* ========================= LẤY GIÁ XE ========================= */ let pricePerKm = vehiclePrices[selectedVehicle]; /* ========================= TÍNH TIỀN ========================= */ let totalPrice = distance * pricePerKm; /* ========================= HIỆN KẾT QUẢ ========================= */ document.getElementById("result") .innerHTML = "Khoảng cách: " + distance.toFixed(2) + " km" + "
" + "Loại xe: " + selectedVehicle + "
" + "Giá ship: " + totalPrice.toFixed(0) + "đ"; /* ========================= TOẠ ĐỘ ROUTE ========================= */ let coords = routeJson.routes[0].geometry.coordinates; let latlngs = coords.map(c => [c[1],c[0]]); /* ========================= XOÁ ROUTE CŨ ========================= */ if(routeLine){ map.removeLayer(routeLine); } /* ========================= VẼ ĐƯỜNG ========================= */ routeLine = L.polyline(latlngs,{ color:'blue', weight:5 }).addTo(map); /* ========================= ZOOM ========================= */ map.fitBounds(routeLine.getBounds()); }
