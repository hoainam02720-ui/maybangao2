async function taodon(){ let pickup = document.getElementById("pickup").value let dropoff = document.getElementById("dropoff").value let loai = document.getElementById("loaixe").value if(!pickup || !dropoff){ alert("Nhập địa chỉ") return } let pick = await timtoado(pickup) let drop = await timtoado(dropoff) let lat1 = pick.lat let lon1 = pick.lon let lat2 = drop.lat let lon2 = drop.lon let distance = getDistanceFromLatLonInKm( lat1, lon1, lat2, lon2 ) let gia = 8000 if(loai=="Xe 4 chỗ"){ gia = 13000 } if(loai=="Xe tải"){ gia = 20000 } let tong = Math.round(distance * gia) let don = { id: Date.now(), pickup, dropoff, loai, distance: distance.toFixed(1), tong, status: "Đang tìm tài xế" } fetch("/taodon",{ method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify(don) }) .then(r=>r.json()) .then(data=>{ document.getElementById("ketqua").innerHTML = `
🚕 ĐƠN ĐẶT XE

${tong.toLocaleString()}đ

📏 Khoảng cách: ${distance.toFixed(1)} km

🚖 ${don.status}
` loadDon() }) } async function timtoado(diachi){ let url = "https://nominatim.openstreetmap.org/search?format=json&q=" + encodeURIComponent(diachi) let res = await fetch(url) let data = await res.json() if(data.length==0){ alert("Không tìm thấy địa chỉ") throw new Error("Lỗi địa chỉ") } return { lat: parseFloat(data[0].lat), lon: parseFloat(data[0].lon) } } function getDistanceFromLatLonInKm( lat1, lon1, lat2, lon2 ){ let R = 6371 let dLat = deg2rad(lat2-lat1) let dLon = deg2rad(lon2-lon1) let a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) * Math.sin(dLon/2) * Math.sin(dLon/2) let c = 2 * Math.atan2( Math.sqrt(a), Math.sqrt(1-a) ) return R * c } function deg2rad(deg){ return deg * (Math.PI/180) }
